# P0-1：棋子世界空间信息面板 — 最终版

## 设计原则

- **所有图标由你提供**（Sprite 资源），代码不硬编码颜色/形状
- **数据驱动，零硬编码**：加新元素/新状态只需在 Inspector 中加一条映射，不改脚本
- 紧凑：面板高度不超过棋子模型的 1.5 倍
- 事件驱动：不每帧 Update

[CreateAssetMenu(menuName = "Chess/Element Icon Map")]
public class ElementIconMap : ScriptableObject
{
    [System.Serializable]
    public struct Entry
    {
        public ElementType element;
        public Sprite icon;
    }
    public Entry[] entries;

    // 运行时查找
    public Sprite GetIcon(ElementType element)
    {
        foreach (var e in entries)
            if (e.element == element) return e.icon;
        return null; // 无匹配时不显示图标
    }
}

### 5. 新建 PieceInfoPanel.cs

**路径**：`Assets/Game/Client/PieceInfoPanel.cs`

这是挂在一个 World Space Canvas Prefab 上的 MonoBehaviour。在 PieceView.BindModel 时 Instantiate 并调用 Initialize。

```csharp
public class PieceInfoPanel : MonoBehaviour
{
    // ---- 配置（Inspector 拖入或从 Resources 加载）----
    [SerializeField] private ElementIconMap elementIconMap;
    [SerializeField] private StatusIconMap statusIconMap;

    // ---- UI 引用（Prefab 上拖好的）----
    [Header("生命值")]
    [SerializeField] private Slider hpBar;
    [SerializeField] private TextMeshProUGUI hpText;

    [Header("属性数值")]
    [SerializeField] private TextMeshProUGUI atkText;
    [SerializeField] private TextMeshProUGUI defText;
    [SerializeField] private TextMeshProUGUI moveText;
    [SerializeField] private TextMeshProUGUI rangeText;

    [Header("元素")]
    [SerializeField] private Image innateElementIcon;    // 所属元素
    [SerializeField] private Image affixedElementIcon;   // 当前附着元素

    [Header("能量")]
    [SerializeField] private Slider energyBar;
    [SerializeField] private Image energyBarFill;        // 填充区 Image（用于变色）

    [Header("装备")]
    [SerializeField] private Image[] equipSlots;         // 长度固定 3

    [Header("状态图标容器")]
    [SerializeField] private Transform statusIconContainer; // 水平 Layout Group
    [SerializeField] private GameObject statusIconPrefab;   // 单个状态图标 Prefab（只有一个 Image 子物体）

    // ---- 运行时 ----
    private PieceModel _model;
    private readonly List<GameObject> _activeStatusIcons = new();

    public void Initialize(PieceModel model)
    {
        _model = model;
        if (elementIconMap == null)
            elementIconMap = Resources.Load<ElementIconMap>("ElementIconMap");
        if (statusIconMap == null)
            statusIconMap = Resources.Load<StatusIconMap>("StatusIconMap");
        RefreshAll();
    }

    // ===== 公开刷新方法（外部在数据变更后调用）=====
    public void RefreshAll() { RefreshHP(); RefreshStats(); RefreshElement(); RefreshEnergy(); RefreshEquipment(); RefreshStatus(); }

    public void RefreshHP()
    {
        if (_model == null) return;
        hpBar.maxValue = _model.MaxHP;
        hpBar.value = _model.CurrentHP;
        hpText.text = $"{_model.CurrentHP}/{_model.MaxHP}";
    }

    public void RefreshStats()
    {
        if (_model == null) return;
        atkText.text = _model.EffectiveAttack.ToString();
        defText.text = _model.EffectiveDefense.ToString();
        moveText.text = _model.MoveRange.ToString();
        rangeText.text = _model.AttackRange.ToString();
    }

    public void RefreshElement()
    {
        if (_model == null || elementIconMap == null) return;

        // 所属元素
        var icon = elementIconMap.GetIcon(_model.Data?.innateElement ?? ElementType.None);
        innateElementIcon.sprite = icon;
        innateElementIcon.gameObject.SetActive(icon != null);

        // 附着元素（Duration > 0 且不是 None）
        bool hasAffixed = _model.AffixedElement != ElementType.None && _model.AffixedElementDuration > 0;
        if (hasAffixed)
        {
            affixedElementIcon.sprite = elementIconMap.GetIcon(_model.AffixedElement);
            affixedElementIcon.gameObject.SetActive(true);
        }
        else
        {
            affixedElementIcon.gameObject.SetActive(false);
        }
    }

    public void RefreshEnergy()
    {
        if (_model?.Energy == null) { energyBar.gameObject.SetActive(false); return; }
        energyBar.gameObject.SetActive(true);
        energyBar.maxValue = _model.Energy.MaxEnergy;
        energyBar.value = _model.Energy.CurrentEnergy;
        // 满能变色
        energyBarFill.color = _model.Energy.IsFull
            ? new Color(1f, 0.84f, 0f)   // 金色 #FFD700
            : new Color(0.27f, 0.53f, 1f); // 蓝色 #4488FF
    }

    public void RefreshEquipment()
    {
        if (_model == null) return;
        for (int i = 0; i < equipSlots.Length; i++)
        {
            var equip = i < _model.EquippedItems.Length ? _model.EquippedItems[i] : null;
            if (equip != null)
            {
                equipSlots[i].sprite = equip.Data?.icon; // 你提供的图标
                equipSlots[i].color = Color.white;
                equipSlots[i].gameObject.SetActive(true);
            }
            else
            {
                equipSlots[i].sprite = null;
                equipSlots[i].color = new Color(0.5f, 0.5f, 0.5f, 0.3f); // 灰色半透明占位
                equipSlots[i].gameObject.SetActive(true); // 始终显示 3 个槽位
            }
        }
    }

    public void RefreshStatus()
    {
        if (statusIconMap == null || statusIconPrefab == null) return;
        if (_model == null) return;

        // 清除旧图标
        foreach (var go in _activeStatusIcons) Destroy(go);
        _activeStatusIcons.Clear();

        // 动态创建当前激活的状态图标（加状态只需在此追加条件，不加脚本字段）
        ShowStatusIf("DefDown", _model.CurrentDefenseReduction > 0);
        ShowStatusIf("DefUp",   _model.TemporaryDefenseBonus > 0);
        ShowStatusIf("Frozen",  _model.IsFrozen);
        // 未来加新状态：加一行 ShowStatusIf("NewStatus", condition);
    }

    private void ShowStatusIf(string key, bool condition)
    {
        if (!condition) return;
        var sprite = statusIconMap.GetIcon(key);
        if (sprite == null) return;
        var go = Instantiate(statusIconPrefab, statusIconContainer);
        go.GetComponent<Image>().sprite = sprite;
        go.SetActive(true);
        _activeStatusIcons.Add(go);
    }

    // ===== 显隐 =====
    public void Show() => gameObject.SetActive(true);
    public void Hide() => gameObject.SetActive(false);
}
```

**重点是**：`RefreshStatus()` 里只有一个 `ShowStatusIf("Key", condition)` 列表。加新状态 = 加一行 + 在 StatusIconMap.asset 中加一条映射。**不需要加新的 Image 字段。**

### 10. Editor 搭建（你在 Unity 中手动做）

**A. 创建资源**

1. `Assets/Game/Resources/ElementIconMap.asset`：右键 → Create → Chess → Element Icon Map，拖入 Fire/Water/Thunder/Ice/None 5 个 Sprite
2. `Assets/Game/Resources/StatusIconMap.asset`：右键 → Create → Chess → Status Icon Map，拖入 DefDown/DefUp/Frozen 3 个 Sprite
3. 每个 EquipmentData.asset 的 icon 字段拖入对应 Sprite
4. PieceView Prefab 的 `pieceInfoPanelPrefab` 字段拖入 PieceInfoPanel Prefab

**B. PieceInfoPanel Prefab 结构**

```
PieceInfoPanel (Canvas, World Space, Scale≈0.008, 宽≈220px)
  └── Billboard (挂 Billboard.cs 脚本，LateUpdate 面朝相机)
       └── Panel (Image, 半透明深灰 rgba(0,0,0,0.55), 圆角)
            ├── TopRow (Horizontal Layout Group, 间距4px)
            │   ├── InnateElementIcon (Image, 22×22, 圆形Mask)
            │   ├── HPBar (Slider, Fill红色, 宽60px)
            │   ├── HPText (TMP, 字号9)
            │   └── AffixedElementIcon (Image, 22×22, 默认隐藏)
            ├── StatsRow (HLG, 间距6px)
            │   ├── AtkText (TMP, 字号9)
            │   ├── DefText (TMP, 字号9)
            │   ├── MoveText (TMP, 字号9)
            │   └── RangeText (TMP, 字号9)
            ├── EnergyBar (Slider, Fill蓝色, 宽180px, 高4px)
            ├── EquipmentRow (HLG, 间距2px)
            │   ├── Slot0 (Image, 24×24)
            │   ├── Slot1 (Image, 24×24)
            │   └── Slot2 (Image, 24×24)
            └── StatusContainer (HLG, 间距2px)
                 (子物体由代码 Instantiate statusIconPrefab 动态生成)
```

**C. statusIconPrefab**

一个简单 Prefab：根对象为空，子对象为 Image(16×16)。Prefab 挂在 PieceInfoPanel 的 `statusIconPrefab` 字段。
